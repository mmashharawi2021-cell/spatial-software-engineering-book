# EX-010

Illustrative excerpt; not a standalone runnable example.

```text
CREATE TABLE app_users (
  id uuid PRIMARY KEY,
  display_name text NOT NULL
);

CREATE TABLE projects (
  id uuid PRIMARY KEY,
  name text NOT NULL,
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE project_members (
  project_id uuid REFERENCES projects(id) ON DELETE CASCADE,
  user_id uuid REFERENCES app_users(id) ON DELETE CASCADE,
  role text NOT NULL CHECK (role IN ('viewer','editor','reviewer','admin')),
  PRIMARY KEY (project_id, user_id)
);

CREATE TABLE features (
  id uuid PRIMARY KEY,
  project_id uuid NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  feature_type text NOT NULL CHECK (feature_type IN ('point','polygon')),
  status text NOT NULL CHECK (status IN ('draft','submitted','reviewed','approved')),
  geom geometry(Geometry, 4326) NOT NULL,
  version integer NOT NULL DEFAULT 1 CHECK (version > 0),
  created_by uuid NOT NULL REFERENCES app_users(id),
  updated_by uuid NOT NULL REFERENCES app_users(id),
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  CHECK (ST_IsValid(geom)),
  CHECK (NOT ST_IsEmpty(geom)),
  CHECK (
    (feature_type='point' AND GeometryType(geom)='POINT') OR
    (feature_type='polygon' AND GeometryType(geom) IN ('POLYGON','MULTIPOLYGON'))
  )
);

CREATE INDEX features_geom_gix ON features USING GIST (geom);
CREATE INDEX features_project_status_idx ON features(project_id, status);
```
