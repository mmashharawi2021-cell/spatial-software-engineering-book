# EX-009

Illustrative excerpt; not a standalone runnable example.

```text
WITH bounds AS (
  SELECT
    ST_TileEnvelope(:z, :x, :y) AS tile_3857,
    ST_Transform(ST_TileEnvelope(:z, :x, :y), 4326) AS query_4326
), mvtgeom AS (
  SELECT
    b.id,
    ST_AsMVTGeom(
      ST_Transform(b.geom, 3857),
      bounds.tile_3857,
      4096, 64, true
    ) AS geom
  FROM buildings AS b
  CROSS JOIN bounds
  WHERE b.geom && bounds.query_4326
    AND ST_Intersects(b.geom, bounds.query_4326)
)
SELECT ST_AsMVT(mvtgeom.*, 'buildings', 4096, 'geom')
FROM mvtgeom;
```
